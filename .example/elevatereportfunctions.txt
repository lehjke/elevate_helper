
Public Sub CopyDataMain(ByRef FileName As String, ByRef Folder As String, ByRef AWT() As Variant, ByRef ATTD() As Variant)

FileName = ActiveSheet.Cells(2, 1).Value
Folder = ActiveSheet.Cells(2, 2).Value

Dim nSteps As Integer: nSteps = ActiveSheet.UsedRange.Rows.Count

ReDim AWT(1 To nSteps - 1)
ReDim ATTD(1 To nSteps - 1)

For i = 2 To nSteps
    AWT(i - 1) = ActiveSheet.Cells(i, 7).Value
    ATTD(i - 1) = ActiveSheet.Cells(i, 11).Value
Next i

ActiveWorkbook.Close Savechanges:=False

End Sub

Public Sub CopyDataSub(ByVal FileName As String, ByVal Folder As String, ByRef JobData() As String, ByRef BuildingData As BuildingDataType, ByRef ElevatorData As ElevatorDataType, ByRef PassengerData As PassengerDataType)

FileName = FileName + ".csv"
Folder = Folder + "\"

Workbooks.Open FileName:=Folder & FileName, Local:=True

Dim JobDataRow As Integer
Dim AnalDataRow As Integer
Dim BuildingDataRow As Integer
Dim ElevatorDataRow As Integer
Dim PassengerDataRow As Integer

For i = 1 To ActiveSheet.UsedRange.Rows.Count

    If ActiveSheet.Cells(i, 1).Value = "JOB DATA" Then
        JobDataRow = i
    End If
    
    If ActiveSheet.Cells(i, 1).Value = "ANALYSIS DATA" Then
        AnalDataRow = i
    End If
    
    If ActiveSheet.Cells(i, 1).Value = "BUILDING DATA" Then
        BuildingDataRow = i
    End If

    If ActiveSheet.Cells(i, 1).Value = "ELEVATOR DATA" Then
        ElevatorDataRow = i
    End If
    
    If ActiveSheet.Cells(i, 1).Value = "PASSENGER DATA" Then
        PassengerDataRow = i
    End If

Next i

'JOB DATA
ReDim JobData(1 To 7)
For i = 1 To 7
    JobData(i) = ActiveSheet.Cells(JobDataRow + i, 4).Value
Next i

'BUILDING DATA
Dim sValue1 As String
Dim sValue2 As String
Dim sLen As Integer

BuildingData.BuildingType = ActiveSheet.Cells(ElevatorDataRow - 8, 6).Value
BuildingData.Absenteeism = ActiveSheet.Cells(ElevatorDataRow - 9, 6).Value

Dim BuildingType As String
If BuildingData.BuildingType = "Office" Then
    BuildingType = "Офис"
ElseIf BuildingData.BuildingType = "Residential" Then
    BuildingType = "Жилье"
ElseIf BuildingData.BuildingType = "Hotel" Then
    BuildingType = "Отель"
Else
    BuildingType = "БКТ"
End If

Dim Absenteeism As Double
Absenteeism = (100 - BuildingData.Absenteeism) / 100

BuildingData.NoFloors = ActiveSheet.Cells(ElevatorDataRow - 2, 6).Value
BuildingData.TotalPeople = 0
BuildingData.cTotalPeople = 0
BuildingData.NoExitFloors = 0

ReDim BuildingData.FloorName(1 To BuildingData.NoFloors)
ReDim BuildingData.FloorHeight(1 To BuildingData.NoFloors)
ReDim BuildingData.FloorLevel(1 To BuildingData.NoFloors)
ReDim BuildingData.FloorType(1 To BuildingData.NoFloors)
ReDim BuildingData.FloorFactor(1 To BuildingData.NoFloors)
ReDim BuildingData.NoPeople(1 To BuildingData.NoFloors)
ReDim BuildingData.EntranceFloor(1 To BuildingData.NoFloors)
ReDim BuildingData.Bias(1 To BuildingData.NoFloors)

Dim tFloorName() As String

For i = 1 To BuildingData.NoFloors

    Erase tFloorName
    
    sLen = Len(ActiveSheet.Cells(BuildingDataRow + i + 1, 1).Value)
    sValue1 = ActiveSheet.Cells(BuildingDataRow + i + 1, 1).Value
    sValue2 = Right(sValue1, sLen - 6)
    'crutch for decimal values
    sValue2 = Replace(sValue2, ".", ",")
    
    BuildingData.FloorName(i) = CDbl(sValue2)
    BuildingData.FloorHeight(i) = ActiveSheet.Cells(BuildingDataRow + i + 1, 3).Value
    
    BuildingData.NoPeople(i) = ActiveSheet.Cells(BuildingDataRow + i + 1, 5).Value
    BuildingData.EntranceFloor(i) = ActiveSheet.Cells(BuildingDataRow + i + 1, 11).Value
    
    If BuildingData.FloorName(i) < 0 Then
        BuildingData.FloorType(i) = "Парковка"
        BuildingData.FloorFactor(i) = 1.2
    ElseIf BuildingData.FloorName(i) > 0 And (BuildingData.EntranceFloor(i) = "Yes") Then
        BuildingData.FloorType(i) = "Лобби"
        BuildingData.FloorFactor(i) = 0
    Else
        BuildingData.FloorType(i) = BuildingType
        BuildingData.FloorFactor(i) = Absenteeism
        BuildingData.NoExitFloors = BuildingData.NoExitFloors + 1
    End If
    
    If BuildingData.EntranceFloor(i) = "Yes" Then
        For j = 1 To BuildingData.NoFloors
            If ActiveSheet.Cells(PassengerDataRow + 15 + j, 1).Value = sValue1 Then
                BuildingData.Bias(i) = ActiveSheet.Cells(PassengerDataRow + 15 + j, 4)
                Exit For
            End If
            
            If ActiveSheet.Cells(PassengerDataRow + 15 + j, 1).Value Like "*&*" Then
                tFloorName = Split(ActiveSheet.Cells(PassengerDataRow + 15 + j, 1).Value, " & ")
                If (tFloorName(0) = sValue1) Or (tFloorName(1) = sValue1) Then
                    BuildingData.Bias(i) = ActiveSheet.Cells(PassengerDataRow + 15 + j, 4)
                    Exit For
                End If
            End If
        Next j
    End If
    
    BuildingData.TotalPeople = BuildingData.TotalPeople + BuildingData.NoPeople(i)
    BuildingData.cTotalPeople = BuildingData.cTotalPeople + BuildingData.NoPeople(i) * BuildingData.FloorFactor(i)
    
Next i

Dim LowerLevel As Double: LowerLevel = 0

If BuildingData.FloorName(1) < 1 Then
    For i = 1 To Abs(BuildingData.FloorName(1))
         LowerLevel = LowerLevel - BuildingData.FloorHeight(i)
    Next i
End If

BuildingData.FloorLevel(1) = LowerLevel

For i = 2 To BuildingData.NoFloors
    BuildingData.FloorLevel(i) = BuildingData.FloorLevel(i - 1) + BuildingData.FloorHeight(i - 1)
Next i

For i = 1 To BuildingData.NoFloors
    If BuildingData.FloorType(i) = "Парковка" Then
        BuildingData.NoPeople(i) = (BuildingData.TotalPeople * BuildingData.Bias(i)) / 120
    End If
Next i

'ELEVATOR DATA
ElevatorData.Dispatcher = ActiveSheet.Cells(AnalDataRow + 3, 6).Value

ElevatorData.NoElevators = 0

For i = 1 To ActiveSheet.UsedRange.Columns.Count
    If ActiveSheet.Cells(ElevatorDataRow + 1, 3 + i).Value <> "" Then
        ElevatorData.NoElevators = ElevatorData.NoElevators + 1
    End If
Next i

ReDim ElevatorData.Spec(1 To ElevatorData.NoElevators, 1 To 10)

For i = 1 To ElevatorData.NoElevators
    For j = 1 To 10
        ElevatorData.Spec(i, j) = ActiveSheet.Cells(ElevatorDataRow + 1 + j, 3 + i).Value
    Next j
Next i



ReDim ElevatorData.FloorsServed(1 To ElevatorData.NoElevators, 1 To BuildingData.NoFloors)

For i = 1 To ElevatorData.NoElevators
    For j = 1 To BuildingData.NoFloors
        ElevatorData.FloorsServed(i, j) = ActiveSheet.Cells(ElevatorDataRow + 14 + j, 3 + i).Value
    Next j
Next i

'PASSENGER DATA

Dim Incoming As String
Dim Outgoing As String
Dim Interfloor As String

Incoming = Replace(ActiveSheet.Cells(PassengerDataRow + 4, 4).Value, ".", ",")
Outgoing = Replace(ActiveSheet.Cells(PassengerDataRow + 5, 4).Value, ".", ",")
Interfloor = Replace(ActiveSheet.Cells(PassengerDataRow + 6, 4).Value, ".", ",")

PassengerData.Incoming = CDbl(Incoming)
PassengerData.Outgoing = CDbl(Outgoing)
PassengerData.Interfloor = CDbl(Interfloor)

ActiveWorkbook.Close Savechanges:=False

End Sub

        End If

    If PassengerData.Incoming = 100 Then
        ActiveWorkbook.Sheets("Оценка").UsedRange.Find("Target WT", LookAt:=xlWhole).Offset(0, 1).Value = 30

Dim nSteps As Integer

    gLimit = 120
    nSteps = 8
ElseIf BuildingData.BuildingType = "Hotel" Then
    gLimit = 80
    nSteps = 13
ElseIf BuildingData.BuildingType = "Office" Then
    gLimit = 80
    nSteps = 13
Else
    gLimit = 80
    nSteps = 13
End If

nStepsFromMain = UBound(AWT)

ReDim Preserve ATTD(1 To nSteps)
ReDim Preserve AIS(1 To nSteps)
ReDim Preserve ALW(1 To nSteps)

For i = nStepsFromMain + 1 To nSteps
    AWT(i) = i * gLimit
    ATTD(i) = i * gLimit
    AIS(i) = i * gLimit
    ALW(i) = i * gLimit
Next i

QuickSort AWT, LBound(AWT), UBound(AWT)
QuickSort ATTD, LBound(ATTD), UBound(ATTD)
QuickSort AIS, LBound(AIS), UBound(AIS)
QuickSort ALW, LBound(ALW), UBound(ALW)

'crutch for LINEST START
Dim lastHC5 As Integer: lastHC5 = nSteps

For i = 1 To nSteps
    If AWT(i) > gLimit Then
        lastHC5 = i - 1
        Exit For
    End If
Next i

Call linest(AIS, lastHC5)
Call linest(ALW, lastHC5)

QuickSort AIS, LBound(AIS), UBound(AIS)
QuickSort ALW, LBound(ALW), UBound(ALW)

'crutch for LINEST END

Dim rCol As Integer: rCol = 0
rCol = ActiveWorkbook.Sheets("Оценка").UsedRange.Find("Record", , , LookAt:=xlWhole).column

Dim mScalseAIS As Integer: mScalseIS = 0
Dim mScalseALW As Integer: mScalseLW = 0
Dim mUnitAIS As Integer: mUnitIS = 0
Dim mUnitALW As Integer: mUnitLW = 0
Dim xScale As Integer: xScale = 0

If BuildingData.BuildingType = "Residential" Then
    xScale = 6
Else
    xScale = 8
End If

For i = 1 To UBound(AWT)

    If AWT(i) < gLimit Then
        ActiveWorkbook.Sheets("Оценка").Cells(7, rCol + i).Value = AWT(i)
        ActiveWorkbook.Sheets("Оценка").Cells(9, rCol + i).Value = ATTD(i)
        ActiveWorkbook.Sheets("Оценка").Cells(11, rCol + i).Value = AIS(i)
        ActiveWorkbook.Sheets("Оценка").Cells(13, rCol + i).Value = ALW(i)
    Else
        Exit For
    End If
    
Next i

mScalseAIS = (Int(AIS(i - 1) / xScale) + 2) * xScale
mUnitAIS = mScalseAIS / xScale

Public Sub csvParse(ByVal FileName As String, ByVal Folder As String, ByRef AIS() As Variant, ByRef ALW() As Variant, ByRef BuildingData As BuildingDataType, ByRef ElevatorData As ElevatorDataType, ByVal step As Integer)
    
    If step < 10 Then
        FileName = Left(FileName, Len(FileName) - 3) & " 0" & CStr(step)
    Else
        FileName = Left(FileName, Len(FileName) - 3) & " " & CStr(step)
    End If
    
    FileName = FileName + ".csv"
    Folder = Folder + "\"

    Workbooks.Open FileName:=Folder & FileName, Local:=True
    
    Dim sRow As Long: sRow = 0
    Dim eRow As Long: eRow = 0
    Dim tPass As Long: tPass = 0
    Dim LW As Double: LW = 0
    
    sRow = ActiveSheet.UsedRange.Find("breakdown", , , xlPart).row + 12
    eRow = ActiveSheet.UsedRange.Find("spatial", , , xlPart).row - 2
    
    For i = sRow To eRow
        tPass = tPass + 1
        If ActiveSheet.Cells(i, 11).Value >= 90 Then
            LW = LW + 1
        End If
    Next i
      
    ALW(step) = 100 * LW / tPass


    
    Dim cRow As Long: cRow = ActiveSheet.UsedRange.Find("spatial", , , xlPart).row + 2
    
    Dim hFloor As Double: hFloor = 0
    Dim hFloorInd As Integer: hFloorInd = 0
    
    Dim sLen As Integer: sLen = 0
    Dim sValue1 As Variant: sValue1 = ""
    Dim sValue2 As Variant: sValue2 = ""
    
    Dim nrTrip As Integer: nrTrip = 0
    Dim sumStop As Integer: sumStop = 0
    Dim nrStop() As Integer
    
    Dim sumAIS As Double: sumALS = 0
    Dim sAIS() As Double
    ReDim sAIS(1 To ElevatorData.NoElevators)
    
    For i = 1 To ElevatorData.NoElevators
    
        sLen = Len(ElevatorData.Spec(i, 5))
        sValue1 = ElevatorData.Spec(i, 5)
        sValue2 = Right(ElevatorData.Spec(i, 5), sLen - 6)
        'crutch for decimal values
        sValue2 = Replace(sValue2, ".", ",")

        hFloor = CDbl(sValue2)
        
        For j = 1 To BuildingData.NoFloors
            If hFloor = BuildingData.FloorName(j) Then
                hFloorInd = j
                Exit For
            End If
        Next j
        
        nrTrip = 0
        sumStop = 0
        ReDim nrStop(1 To 1)

        Do While (ActiveSheet.Cells(cRow, 1) = i) And (ActiveSheet.Cells(cRow + 1, 1) = i)

            If (ActiveSheet.Cells(cRow, 3).Value = hFloorInd) And (ActiveSheet.Cells(cRow + 1, 3).Value = hFloorInd) Then
                    nrTrip = nrTrip + 1
                    ReDim Preserve nrStop(1 To nrTrip)
                    nrStop(nrTrip) = 0
                    
                    Do While (ActiveSheet.Cells(cRow + 2, 3).Value <> hFloorInd) And (ActiveSheet.Cells(cRow + 2, 1) = i)
                        
                        If ActiveSheet.Cells(cRow + 2, 3) <> ActiveSheet.Cells(cRow + 3, 3) Then
                            nrStop(nrTrip) = nrStop(nrTrip) + 1
                        End If
                        cRow = cRow + 1
                    Loop
                    
            End If
            cRow = cRow + 1
            
        Loop
        cRow = cRow + 1
        
        For x = 1 To UBound(nrStop)
            sumStop = sumStop + nrStop(x)
        Next x
        
        If nrTrip Then
            sAIS(i) = sumStop / nrTrip - 1
        Else
            sAIS(i) = 0
        End If
        
    Next i
        
    For x = 1 To UBound(sAIS)
        sumAIS = sumAIS + sAIS(x)
    Next x
    
    AIS(step) = sumAIS / UBound(sAIS)
    
    ActiveWorkbook.Close Savechanges:=False

End Sub

Sub QuickSort(arr() As Variant, ByVal low As Long, ByVal high As Long)
    Dim pivot As Variant
    Dim i As Long, j As Long
    Dim temp As Variant
    
    If low < high Then
        pivot = arr((low + high) \ 2)
        i = low
        j = high
        
        Do While i <= j
            Do While arr(i) < pivot And i < high
                i = i + 1
            Loop
            Do While arr(j) > pivot And j > low
                j = j - 1
            Loop
            
            If i <= j Then
                temp = arr(i)
                arr(i) = arr(j)
                arr(j) = temp
                i = i + 1
                j = j - 1
            End If
        Loop
        
        If low < j Then QuickSort arr, low, j
        If i < high Then QuickSort arr, i, high
    End If
End Sub

Function linest(ByRef data() As Variant, lastHC5 As Integer)
    
    Dim linestResult As Variant
    
    Dim aX() As Variant
    Dim aY() As Variant
    
    Dim x4 As Double: x4 = 0
    Dim x3 As Double: x3 = 0
    Dim x2 As Double: x2 = 0
    Dim x1 As Double: x1 = 0
    Dim x0 As Double: x0 = 0
    
    ReDim aX(1 To 4, 1 To lastHC5)
    For i = 1 To lastHC5
        aX(1, i) = i
        aX(2, i) = i ^ 2
        aX(3, i) = i ^ 3
        aX(4, i) = i ^ 4
    Next i
    
    ReDim aY(1 To lastHC5)
    For i = 1 To lastHC5
        aY(i) = data(i)
    Next i
    
    linestResult = Application.WorksheetFunction.linest(aY, aX, True, False)
    x4 = linestResult(1)
    x3 = linestResult(2)
    x2 = linestResult(3)
    x1 = linestResult(4)
    x0 = linestResult(5)
    
    For i = 1 To lastHC5
        data(i) = x4 * i ^ 4 + x3 * i ^ 3 + x2 * i ^ 2 + x1 * i + x0
        
        If data(i) < 0.1 Then
            data(i) = 0
        End If
    Next i
    
End Function
